version: "3.9"

services:
  mariadb:
    image: mariadb:10.11
    restart: unless-stopped
    command: ["mariadbd", "--event-scheduler=ON"]
    environment:
      MARIADB_ROOT_PASSWORD: rootpassword
      MARIADB_DATABASE: Marais_R_Site
      MARIADB_USER: Marais_R_Site_User
      MARIADB_PASSWORD: Marais_R_Site_User/123
    volumes:
      - vol_BDD:/var/lib/mysql
    networks:
      - marais_net

  phpmyadmin:
    image: phpmyadmin/phpmyadmin
    restart: unless-stopped
    ports:
      - "127.0.0.1:8080:80"
    environment:
      PMA_HOST: mariadb
      PMA_ARBITRARY: 1
    depends_on:
      - mariadb
    networks:
      - marais_net

  grafana:
    image: grafana/grafana
    restart: unless-stopped
    ports:
      - "127.0.0.1:3000:3000"
    volumes:
      - vol_grafana:/var/lib/grafana
    networks:
      - marais_net

volumes:
  vol_BDD:
  vol_grafana:

networks:
  marais_net:
    driver: bridge
